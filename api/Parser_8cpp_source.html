295 std::unique_ptr<ASTNode> Parser::ParseBinOpRHS(int ExprPrec, std::unique_ptr<ASTNode> LHS)
296 {
297 while (true)
298 {
299 int TokPrec = GetTokPrecedence();
300
301 if (TokPrec < ExprPrec)
302 return LHS;
303

420 {
421 if (CurTok != tok_import)
422 return LogErrorExpected("import", "to start import statement");
423
424 getNextToken();
425 if (CurTok != tok_identifier)
426 return LogErrorExpected("module name", "after 'import'");
427
428 std::string moduleName = lexer.IdentifierStr;
429 getNextToken();
430 if (CurTok != ';')
431 return LogErrorExpected("';'", "after import statement");
432 getNextToken();
433 // To avoid duplicate imports
434 if (AlreadyImported.count(moduleName))

436 return std::make_unique<ImportAST>(moduleName, std::vector<std::unique_ptr<ASTNode>>{});
437 }
438 AlreadyImported.insert(moduleName);
439
440 std::string importPath = "lib/" + moduleName + ".rvt";
441
442 // This fallback is done so that the compiler could pick up the file whether rivet is executed from the repo root or from the build directory.
443 if (!std::filesystem::exists(importPath))
444 {
445 const std::string fallbackPath = "../lib/" + moduleName + ".rvt";
446 if (std::filesystem::exists(fallbackPath))
447 {
448 importPath = fallbackPath;
449 }
450 }
451
452 Lexer importLexer(importPath);
453 Parser importParser(importLexer);
454
455 std::cout << "Injecting source from module:" << moduleName << std::endl;
456 auto importedAST = importParser.ParseFile();

465 std::unique_ptr<ASTNode> Parser::ParseForStatement()
466 {
467 // Consume 'for'
468 getNextToken();
469
470 // Consume '('
471 if (CurTok != '(')
472 return LogErrorExpected("(", "after 'for'");
473 getNextToken();
474
475 // Iterator variable name
476 if (CurTok != tok_identifier)
477 return LogErrorExpected("identifier", "after '('");
478
479 std::string varName = lexer.IdentifierStr;
480 getNextToken();
481
482 // Consume 'in'
483 if (CurTok != tok_in)
484 return LogErrorExpected("'in'", "after iterator variable");
485 getNextToken();
486
487 // Start expression
488 auto start = ParseExpression();
489 if (!start)
490 return nullptr;
491
492 // Consume 'to'
493 if (CurTok != tok_to)
494 return LogErrorExpected("'to'", "after start expression");
495 getNextToken();
496
497 // End expression
498 auto end = ParseExpression();
499 if (!end)
500 return nullptr;
501
502 // Optional Step
503 std::unique_ptr<ASTNode> step = nullptr;
504 if (CurTok == tok_step)
505 {
506 getNextToken();
507 step = ParseExpression();
508 if (!step)
509 return nullptr;
510 }
511
512 // Consume ')'
513 if (CurTok != ')')
514 return LogErrorExpected(")", "at end of 'for'");
515
516 getNextToken();
517
518 // Loop Body
519 auto body = ParseBlock();
520 if (!body)
521 return nullptr;
522
523 return std::make_unique<ForAST>(varName, std::move(start), std::move(end), std::move(step), std::move(body));
524 }
