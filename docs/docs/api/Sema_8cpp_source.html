79
80 bool VariableDeclAST::typeCheck(SymbolTable& symTab)
81 {
82 BaseType declaredBase = BaseType::Unknown;
83 if (Type == "int")
84 declaredBase = BaseType::Int;
85 else if (Type == "str")
86 declaredBase = BaseType::String;
87
88 if (declaredBase == BaseType::Unknown)
89 {
90 std::cerr << "Semantic Error: Unknown declared type '" << Type << "' for variable '" << Name << "'.\n";
91 return false;
92 }
93
94 TypeInfo declaredType(declaredBase, IsRef, ArrayCapacity, IsOptRef);
95
96 if (IsRef && !IsOptRef && !InitVal)
97 {
98 std::cerr << "Semantic Error: Reference variable '" << Name << "' must be initialized.\n";
99 return false;
100 }
101
102 if (InitVal)
103 {
104 if (!InitVal->typeCheck(symTab))
105 return false;
106
107 if (InitVal->ExprType != declaredType)
108 {
109 bool allowed = false;
110 // Allow initializing optref with a strict ref of the same base type
111 if (declaredType.Base == InitVal->ExprType.Base &&
112 declaredType.IsRef && InitVal->ExprType.IsRef &&
113 declaredType.IsOptRef && !InitVal->ExprType.IsOptRef &&
114 declaredType.ArrayCapacity == InitVal->ExprType.ArrayCapacity)
115 {
116 allowed = true;
117 }
118 // Allow initializing any optref with the null literal
119 if (!allowed && declaredType.IsOptRef && InitVal->ExprType.IsOptRef)
120 {
121 allowed = true;
122 }
123
124 if (!allowed)
125 {
126 std::cerr << "Semantic Error: Type mismatch in declaration of '" << Name
127 << "'. Expected " << declaredType.toString()
128 << ", found " << InitVal->ExprType.toString() << ".\n";
129 return false;
130 }
131 }
132 }
133
134 if (!symTab.insert(Name, declaredType, InitVal != nullptr))
135 {
136 std::cerr << "Semantic Error: Variable '" << Name << "' already declared in this scope.\n";
137 return false;
138 }
139
140 ExprType = declaredType;
141 return true;
142 }

148 bool BinaryOpAST::typeCheck(SymbolTable& symTab)
149 {
150 if (!LHS->typeCheck(symTab) || !RHS->typeCheck(symTab))
151 return false;
152
153 if (Op == '=')
154 {
155 bool isVariable = dynamic_cast<VariableAST*>(LHS.get()) != nullptr;
156 bool isDeref = dynamic_cast<DerefAST*>(LHS.get()) != nullptr;
157 bool isIndex = dynamic_cast<IndexAST*>(LHS.get()) != nullptr;
158
159 if (!isVariable && !isDeref && !isIndex)
160 {
161 std::cerr << "Semantic Error: Left Hand Side of assignment must be a variable, dereference, or array index.\n";
162 return false;
163 }
164 if (LHS->ExprType != RHS->ExprType)
165 {
166 bool allowed = false;
167 // Allow assigning a strict ref to an optref of the same base
168 if (LHS->ExprType.Base == RHS->ExprType.Base &&
169 LHS->ExprType.IsRef && RHS->ExprType.IsRef &&
170 LHS->ExprType.IsOptRef && !RHS->ExprType.IsOptRef &&
171 LHS->ExprType.ArrayCapacity == RHS->ExprType.ArrayCapacity)
172 {
173 allowed = true;
174 }
175 // Allow assigning the null literal to any optref
176 if (!allowed && LHS->ExprType.IsOptRef && RHS->ExprType.IsOptRef)
177 {
178 allowed = true;
179 }
180
181 if (!allowed)
182 {
183 std::cerr << "Semantic Error: Types of LHS and RHS must be the same.\n";
184 return false;
185 }
186 }
187 ExprType = LHS->ExprType;
188 return true;
189 }
190
191 if (LHS->ExprType.Base != BaseType::Int || RHS->ExprType.Base != BaseType::Int ||
192 LHS->ExprType.IsRef || RHS->ExprType.IsRef ||
193 LHS->ExprType.isArray() || RHS->ExprType.isArray())
194 {
195 std::cerr << "Semantic Error: Binary operators require integer operands.\n";
196 return false;
197 }
198
199 ExprType = TypeInfo(BaseType::Int, false);
200 return true;
201 }

231 bool BlockAST::typeCheck(SymbolTable& symTab)
232 {
233 symTab.enterScope();
234
235 // Save the caller's has-return state; bubble ours up afterwards.
236 bool savedHasReturn = CurrentFunctionHasReturn;
237 CurrentFunctionHasReturn = false;
238
239 bool hasReturnInBlock = false;
240 for (auto& stmt : Statements)
241 {
242 if (!stmt->typeCheck(symTab))
243 {
244 symTab.exitScope();
245 CurrentFunctionHasReturn = savedHasReturn || hasReturnInBlock;
246 return false;
247 }
248 if (CurrentFunctionHasReturn)
249 hasReturnInBlock = true;
250 }
251
252 symTab.exitScope();
253 // Propagate: outer scope sees a return if this block or any inner block had one.
254 CurrentFunctionHasReturn = savedHasReturn || hasReturnInBlock;
255 ExprType = TypeInfo(BaseType::Int, false);
256 return true;
257 }

322 bool CallAST::typeCheck(SymbolTable& symTab)
323 {
324 // Look up purely from the semantic FunctionRegistry — no LLVM module access.
325 auto it = symTab.FunctionRegistry.find(Callee);
326 if (it == symTab.FunctionRegistry.end())
327 {
328 std::cerr << "Semantic Error: Unknown function '" << Callee << "'.\n";
329 return false;
330 }
331
332 const FunctionSignature& Sig = it->second;
333
334 if (Sig.Params.size() != Args.size())
335 {
336 std::cerr << "Semantic Error: Incorrect number of arguments passed to function '"
337 << Callee << "'. Expected " << Sig.Params.size()
338 << ", got " << Args.size() << ".\n";
339 return false;
340 }
341
342 for (size_t i = 0; i < Args.size(); ++i)
343 {
344 if (!Args[i]->typeCheck(symTab))
345 return false;
346
347 const TypeInfo& Expected = Sig.Params[i];
348 const TypeInfo& Actual = Args[i]->ExprType;
349
350 bool compatible = (Expected == Actual);
351 // Allow passing a strict ref where an optref is expected
352 if (!compatible && Expected.Base == Actual.Base &&
353 Expected.IsRef && Actual.IsRef &&
354 Expected.IsOptRef && !Actual.IsOptRef)
355 compatible = true;
356
357 if (!compatible)
358 {
359 std::cerr << "Semantic Error: Type mismatch for argument " << i
360 << " in call to '" << Callee << "'. Expected "
361 << Expected.toString() << ", found " << Actual.toString() << ".\n";
362 return false;
363 }
364 }
365
366 ExprType = Sig.ReturnType;
367 return true;
368 }

374 bool ReturnAST::typeCheck(SymbolTable& symTab)
375 {
376 if (FunctionReturnTypeStack.empty())
377 {
378 std::cerr << "Semantic Error: 'return' used outside of a function.\n";
379 return false;
380 }
381
382 TypeInfo expected = FunctionReturnTypeStack.back();
383 if (!RetVal)
384 {
385 if (expected.Base != BaseType::Void)
386 {
387 std::cerr << "Semantic Error: Non-void function must return a value.\n";
388 return false;
389 }
390 ExprType = TypeInfo(BaseType::Void, false);
391 return true;
392 }
393
394 if (!RetVal->typeCheck(symTab))
395 return false;
396
397 if (expected.Base == BaseType::Void)
398 {
399 std::cerr << "Semantic Error: Void function cannot return a value.\n";
400 return false;
401 }
402 if (RetVal->ExprType != expected)
403 {
404 std::cerr << "Semantic Error: Return type mismatch. Expected "
405 << expected.toString() << ", found " << RetVal->ExprType.toString() << ".\n";
406 return false;
407 }
408 ExprType = expected;
409 // Signal to FunctionAST that this code path reaches a return.
410 CurrentFunctionHasReturn = true;
411 return true;
412 }

418 bool FunctionAST::registerSignature(SymbolTable& symTab)
419 {
420 auto toTypeInfo = [&](const std::string& typeName, bool isRef = false, bool isOptRef = false) -> TypeInfo
421 {
422 if (typeName == "int") return TypeInfo(BaseType::Int, isRef, 0, isOptRef);
423 if (typeName == "str") return TypeInfo(BaseType::String, isRef, 0, isOptRef);
424 if (typeName == "void") return TypeInfo(BaseType::Void, false);
425 return TypeInfo(BaseType::Unknown, false);
426 };
427
428 TypeInfo RetInfo = toTypeInfo(ReturnType, false);
429 if (RetInfo.Base == BaseType::Unknown)
430 {
431 std::cerr << "Semantic Error: Unknown return type '" << ReturnType << "' in function '" << Name << "'.\n";
432 return false;
433 }
434
435 std::vector<TypeInfo> ParamTypes;
436 ParamTypes.reserve(Params.size());
437 for (const auto& P : Params)
438 {
439 TypeInfo TI = toTypeInfo(P.TypeName, P.IsRef, P.IsOptRef);
440 if (TI.Base == BaseType::Unknown)
441 {
442 std::cerr << "Semantic Error: Unknown parameter type '" << P.TypeName << "' in function '" << Name << "'.\n";
443 return false;
444 }
445 if (TI.Base == BaseType::Void)
446 {
447 std::cerr << "Semantic Error: Function parameters cannot be void in function '" << Name << "'.\n";
448 return false;
449 }
450 ParamTypes.push_back(TI);
451 }
452
453 symTab.FunctionRegistry[Name] = FunctionSignature{RetInfo, ParamTypes};
454 return true;

457 bool FunctionAST::typeCheck(SymbolTable& symTab)
458 {
459 auto toTypeInfo = [&](const std::string& typeName, bool isRef = false, bool isOptRef = false) -> TypeInfo
460 {
461 if (typeName == "int")
462 return TypeInfo(BaseType::Int, isRef, 0, isOptRef);
463 if (typeName == "str")
464 return TypeInfo(BaseType::String, isRef, 0, isOptRef);
465 if (typeName == "void")
466 return TypeInfo(BaseType::Void, false);
467 return TypeInfo(BaseType::Unknown, false);
468 };
469
470 TypeInfo RetInfo = toTypeInfo(ReturnType, false);
471 if (RetInfo.Base == BaseType::Unknown)
472 {
473 std::cerr << "Semantic Error: Unknown return type '" << ReturnType << "' in function '" << Name << "'.\n";
474 return false;
475 }
476
477 // Build the parameter TypeInfo list and validate each one.
478 std::vector<TypeInfo> ParamTypes;
479 ParamTypes.reserve(Params.size());
480 for (const auto& P : Params)
481 {
482 TypeInfo TI = toTypeInfo(P.TypeName, P.IsRef, P.IsOptRef);
483 if (TI.Base == BaseType::Unknown)
484 {
485 std::cerr << "Semantic Error: Unknown parameter type '" << P.TypeName << "' in function '" << Name << "'.\n";
486 return false;
487 }
488 if (TI.Base == BaseType::Void)
489 {
490 std::cerr << "Semantic Error: Function parameters cannot be void in function '" << Name << "'.\n";
491 return false;
492 }
493 ParamTypes.push_back(TI);
494 }
495
496 // Register signature into FunctionRegistry — NO LLVM IR created here.
497 symTab.FunctionRegistry[Name] = FunctionSignature{RetInfo, ParamTypes};
498
499 symTab.enterScope();
500 for (size_t i = 0; i < Params.size(); ++i)
501 {
502 if (!symTab.insert(Params[i].Name, ParamTypes[i], true))
503 {
504 std::cerr << "Semantic Error: Duplicate parameter name '" << Params[i].Name << "' in function '" << Name << "'.\n";
505 symTab.exitScope();
506 return false;
507 }
508 }
509
510 FunctionReturnTypeStack.push_back(RetInfo);
511 // Reset the flag before checking the body of this function.
512 CurrentFunctionHasReturn = false;
513 bool BodyOk = Body->typeCheck(symTab);
514 bool bodyHasReturn = CurrentFunctionHasReturn;
515 FunctionReturnTypeStack.pop_back();
516 symTab.exitScope();
517
518 if (BodyOk && RetInfo.Base != BaseType::Void && !bodyHasReturn)
519 {
520 std::cerr << "Semantic Error: Non-void function '" << Name
521 << "' does not always return a value.\n";
522 return false;
523 }
524
525 ExprType = TypeInfo(BaseType::Void, false);
526 return BodyOk;
527 }

579
580 bool IndexAST::typeCheck(SymbolTable& symTab)
581 {
582 // Verify the variable exists in symbol table
583 Symbol* sym = symTab.lookup(ArrayName);
584 if (!sym)
585 {
586 std::cerr << "Semantic Error: Array '" << ArrayName << "' is not declared.\n";
587 return false;
588 }
589
590 // Verify it's an array
591 if (!sym->Type.isArray())
592 {
593 std::cerr << "Semantic Error: Variable '" << ArrayName << "' is not an array.\n";
594 return false;
595 }
596
597 // Verify the index expression is valid and resolves to an integer
598 if (!IndexExpr->typeCheck(symTab))
599 return false;
600 if (IndexExpr->ExprType.Base != BaseType::Int || IndexExpr->ExprType.IsRef || IndexExpr->ExprType.isArray())
601 {
602 std::cerr << "Semantic Error: Array '" << ArrayName << "' must be of type int.\n";
603 return false;
604 }
605
606 // Compile-time bounds check: if the index is a hardcoded literal
607 if (auto* NumNode = dynamic_cast<NumberAST*>(IndexExpr.get()))
608 {
609 int indexValue = NumNode->getVal();
610 if (indexValue < 0 || indexValue >= sym->Type.ArrayCapacity)
611 {
612 std::cerr << "Semantic Error: Array index out of bounds. '" << ArrayName
613 << "' has capacity " << sym->Type.ArrayCapacity
614 << ", but accessed at index " << indexValue << ".\n";
615 return false;
616 }
617 }
618
619 // Indexing int[N] yields int
620 ExprType = TypeInfo(sym->Type.Base, false);
621 return true;
622 }
