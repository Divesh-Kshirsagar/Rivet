310 // =========================================================================
311 // THE RIVET DRIVER: Flexible Bare-Metal Linking
312 //
313 // Activates for any target triple that contains "-none-" (the standard
314 // bare-metal OS component in LLVM/GNU triples). WASM is excluded because
315 // it uses its own module ABI — GNU linker scripts do not apply.
316 //
317 // All values flow from the user-supplied --target / --mcpu / --flash-* /
318 // --ram-* flags. Nothing is hardcoded.
319 // =========================================================================
320 // Helper: parse a memory value string that may carry a K/M suffix or be hex.
321 // e.g. "128K" -> 131072, "512K" -> 524288, "0x20000" -> 131072, "65536" -> 65536
322 // This is needed because K/M suffixes are linker-script syntax, NOT assembler syntax.
323 auto parseMemValue = [] ( const std::string &s ) -> uint64_t
324 {
325 if ( s.empty () )
326 return 0;
327 char suffix = s.back ();
328 if ( suffix == 'K' || suffix == 'k' )
329 return std::stoull ( s.substr ( 0, s.size () - 1 ), nullptr, 0 ) * 1024ULL;
330 if ( suffix == 'M' || suffix == 'm' )
331 return std::stoull ( s.substr ( 0, s.size () - 1 ), nullptr, 0 ) * 1024ULL * 1024ULL;
332 return std::stoull ( s, nullptr, 0 ); // plain decimal or 0x hex
333 };
334
335 // Pre-compute the initial stack pointer (top of RAM) as a single hex constant.
336 // This must be a plain number in the .word directive — the GNU assembler does
337 // NOT understand the K/M suffixes that are valid in linker scripts.
338 uint64_t ramOriginVal = parseMemValue ( ramOrigin );
339 uint64_t ramSizeVal = parseMemValue ( ramSize );
340 uint64_t stackTopVal = ramOriginVal + ramSizeVal;
341 std::ostringstream stackTopStream;
342 stackTopStream << "0x" << std::hex << std::uppercase << stackTopVal;
343 std::string stackTop = stackTopStream.str ();
